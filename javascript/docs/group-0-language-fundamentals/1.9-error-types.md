# Error Types & Try/Catch Mechanics

> "Errors happen — know how to handle them gracefully."

---

## 1. Concept Overview

JavaScript's error handling uses the `try/catch/finally` mechanism with various built-in error types. Understanding errors is crucial because:
- Proper error handling prevents crashes
- Different error types indicate different problems
- Error handling affects code reliability
- It's essential for debugging and production code

## 2. Built-in Error Types

### 2.1 Error (Base Class)

```javascript
// Generic error
throw new Error("Something went wrong");

// With try/catch
try {
  throw new Error("Custom error");
} catch (e) {
  console.log(e.message);  // "Custom error"
  console.log(e.name);     // "Error"
  console.log(e.stack);     // Stack trace
}
```

### 2.2 TypeError

Thrown when a value is not of the expected type:

```javascript
// Common cases
let obj = null;
obj.property;  // TypeError: Cannot read property 'property' of null

let num = 42;
num();  // TypeError: num is not a function

let arr = [1, 2, 3];
arr.push.call(null, 4);  // TypeError: Cannot convert undefined or null to object
```

### 2.3 ReferenceError

Thrown when referencing a non-existent variable:

```javascript
// Undeclared variable
console.log(undeclaredVar);  // ReferenceError: undeclaredVar is not defined

// In strict mode
"use strict";
undeclaredVar = 5;  // ReferenceError (vs global in non-strict)
```

### 2.4 SyntaxError

Thrown when code has syntax errors:

```javascript
// Invalid syntax
eval("let x = ;");  // SyntaxError

// JSON parsing
JSON.parse("{ invalid }");  // SyntaxError: Unexpected token
```

### 2.5 RangeError

Thrown when a value is outside allowed range:

```javascript
// Array length
let arr = new Array(-1);  // RangeError: Invalid array length

// Number methods
(42).toFixed(-1);  // RangeError: toFixed() digits argument must be between 0 and 100
(42).toPrecision(0);  // RangeError
```

### 2.6 URIError

Thrown by URI encoding/decoding methods:

```javascript
decodeURIComponent("%");  // URIError: URI malformed
encodeURI("\uD800");      // URIError: URI malformed
```

## 3. Try/Catch/Finally

### 3.1 Basic Structure

```javascript
try {
  // Code that might throw
  riskyOperation();
} catch (error) {
  // Handle error
  console.error("Error:", error.message);
} finally {
  // Always executes
  cleanup();
}
```

### 3.2 Finally Block Behavior

```javascript
function test() {
  try {
    return "try";
  } catch (e) {
    return "catch";
  } finally {
    return "finally";  // This return overrides!
  }
}

test();  // "finally"

// Even with throw
function test2() {
  try {
    throw new Error("test");
  } catch (e) {
    throw e;  // Re-throw
  } finally {
    return "finally";  // Overrides the throw!
  }
}

test2();  // "finally" (no error thrown!)
```

## 4. Error Object Properties

### 4.1 Standard Properties

```javascript
try {
  throw new Error("Test error");
} catch (e) {
  console.log(e.name);      // "Error"
  console.log(e.message);   // "Test error"
  console.log(e.stack);     // Stack trace string
  console.log(e.toString()); // "Error: Test error"
}
```

### 4.2 Custom Error Properties

```javascript
class CustomError extends Error {
  constructor(message, code) {
    super(message);
    this.name = "CustomError";
    this.code = code;
  }
}

try {
  throw new CustomError("Failed", 500);
} catch (e) {
  console.log(e.code);  // 500
  console.log(e instanceof CustomError);  // true
}
```

## 5. Error Handling Patterns

### 5.1 Specific Error Handling

```javascript
try {
  riskyOperation();
} catch (e) {
  if (e instanceof TypeError) {
    // Handle type errors
    console.error("Type error:", e.message);
  } else if (e instanceof ReferenceError) {
    // Handle reference errors
    console.error("Reference error:", e.message);
  } else {
    // Handle other errors
    console.error("Unknown error:", e.message);
  }
}
```

### 5.2 Error Propagation

```javascript
function inner() {
  throw new Error("Inner error");
}

function middle() {
  try {
    inner();
  } catch (e) {
    // Re-throw with context
    throw new Error(`Middle layer: ${e.message}`);
  }
}

function outer() {
  try {
    middle();
  } catch (e) {
    console.error(e.message);  // "Middle layer: Inner error"
  }
}
```

### 5.3 Error Logging

```javascript
function logError(error, context = {}) {
  const errorInfo = {
    message: error.message,
    stack: error.stack,
    name: error.name,
    context: context,
    timestamp: new Date().toISOString()
  };
  
  // Send to logging service
  console.error("Error logged:", errorInfo);
  // fetch("/api/errors", { method: "POST", body: JSON.stringify(errorInfo) });
}

try {
  riskyOperation();
} catch (e) {
  logError(e, { userId: 123, action: "processPayment" });
  throw e;  // Re-throw if needed
}
```

## 6. Custom Error Classes

### 6.1 Extending Error

```javascript
class ValidationError extends Error {
  constructor(message, field) {
    super(message);
    this.name = "ValidationError";
    this.field = field;
  }
}

class NetworkError extends Error {
  constructor(message, statusCode) {
    super(message);
    this.name = "NetworkError";
    this.statusCode = statusCode;
  }
}

// Usage
function validateUser(user) {
  if (!user.email) {
    throw new ValidationError("Email required", "email");
  }
  if (!user.email.includes("@")) {
    throw new ValidationError("Invalid email format", "email");
  }
}
```

### 6.2 Error Hierarchy

```javascript
class AppError extends Error {
  constructor(message, code) {
    super(message);
    this.name = "AppError";
    this.code = code;
  }
}

class DatabaseError extends AppError {
  constructor(message) {
    super(message, "DB_ERROR");
    this.name = "DatabaseError";
  }
}

class ConnectionError extends DatabaseError {
  constructor(message) {
    super(message);
    this.name = "ConnectionError";
  }
}
```

## 7. Async Error Handling

### 7.1 Promises

```javascript
// Promise rejection
Promise.reject(new Error("Failed"))
  .catch(error => {
    console.error("Caught:", error.message);
  });

// Async/await
async function fetchData() {
  try {
    const response = await fetch("/api/data");
    if (!response.ok) {
      throw new Error(`HTTP ${response.status}`);
    }
    return await response.json();
  } catch (error) {
    console.error("Fetch error:", error);
    throw error;  // Re-throw or handle
  }
}
```

### 7.2 Unhandled Rejections

```javascript
// Global handler
process.on("unhandledRejection", (reason, promise) => {
  console.error("Unhandled rejection:", reason);
  // Log or handle
});

// In browser
window.addEventListener("unhandledrejection", event => {
  console.error("Unhandled rejection:", event.reason);
  event.preventDefault();  // Prevent console error
});
```

## 8. Real-World Patterns

### 8.1 Defensive Programming

```javascript
function safeDivide(a, b) {
  if (typeof a !== "number" || typeof b !== "number") {
    throw new TypeError("Arguments must be numbers");
  }
  if (b === 0) {
    throw new RangeError("Division by zero");
  }
  return a / b;
}

// Or return null/undefined instead of throwing
function safeDivideOptional(a, b) {
  try {
    return safeDivide(a, b);
  } catch (e) {
    console.warn("Division failed:", e.message);
    return null;
  }
}
```

### 8.2 Error Boundaries (React-like)

```javascript
class ErrorBoundary {
  constructor(component) {
    this.component = component;
  }
  
  render() {
    try {
      return this.component.render();
    } catch (error) {
      return this.fallback(error);
    }
  }
  
  fallback(error) {
    return {
      type: "ErrorDisplay",
      props: { error: error.message }
    };
  }
}
```

### 8.3 Retry Logic

```javascript
async function retryOperation(operation, maxRetries = 3) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await operation();
    } catch (error) {
      if (i === maxRetries - 1) throw error;
      console.log(`Retry ${i + 1}/${maxRetries}`);
      await new Promise(resolve => setTimeout(resolve, 1000 * (i + 1)));
    }
  }
}
```

## 10. Common Pitfalls & Best Practices

### Pitfall 1: Swallowing Errors

```javascript
// ❌ Problematic
try {
  riskyOperation();
} catch (e) {
  // Silent failure - no logging, no handling
}

// ✅ Better
try {
  riskyOperation();
} catch (e) {
  console.error("Operation failed:", e);
  // Handle appropriately or re-throw
}
```

### Pitfall 2: Catching Too Broadly

```javascript
// ❌ Problematic
try {
  // Many operations
} catch (e) {
  // Handles all errors the same way
}

// ✅ Better
try {
  // Specific operation
} catch (e) {
  if (e instanceof SpecificError) {
    // Handle specifically
  } else {
    throw e;  // Re-throw unknown errors
  }
}
```

### Pitfall 3: Finally Return Override

```javascript
// ❌ Problematic (if unexpected)
function getValue() {
  try {
    return "try";
  } finally {
    return "finally";  // Overrides try return!
  }
}

// ✅ Better: Be aware of finally behavior
function getValue() {
  let value = "default";
  try {
    value = "try";
  } finally {
    // Don't return in finally unless intentional
  }
  return value;
}
```

### Best Practice: Error Handling Strategy

```javascript
// 1. Use specific error types
// 2. Add context to errors
// 3. Log errors appropriately
// 4. Handle errors at the right level
// 5. Don't swallow errors silently
// 6. Use finally for cleanup
// 7. Re-throw when you can't handle
```
