# Type Detection Patterns in JavaScript

> "Knowing what you're working with — reliable type checking strategies."

---

## 1. Concept Overview

JavaScript provides multiple ways to detect types, each with different strengths and weaknesses:
- `typeof` operator — quick but limited
- `instanceof` operator — checks prototype chain
- `Array.isArray()` — specific array check
- `Object.prototype.toString()` — most reliable for built-ins
- Custom type guards — for application-specific types

Understanding type detection is crucial because:
- Different methods work for different scenarios
- Some methods have quirks and edge cases
- Choosing the right method affects code reliability
- It's a common interview topic

## 2. typeof Operator

### 2.1 Basic Usage

```javascript
typeof "hello";      // "string"
typeof 42;            // "number"
typeof 42n;           // "bigint"
typeof true;          // "boolean"
typeof undefined;     // "undefined"
typeof Symbol();      // "symbol"
typeof null;          // "object" (quirk!)
typeof {};            // "object"
typeof [];            // "object"
typeof function(){};  // "function"
```

### 2.2 Limitations

```javascript
// Can't distinguish object types
typeof [];            // "object" (not "array")
typeof null;          // "object" (not "null")
typeof new Date();    // "object"
typeof /regex/;       // "object"

// Functions are detected
typeof function(){};  // "function"
typeof class {};      // "function"
```

## 3. instanceof Operator

### 3.1 Basic Usage

```javascript
[] instanceof Array;           // true
{} instanceof Object;          // true
new Date() instanceof Date;    // true
/regex/ instanceof RegExp;     // true

// Works with classes
class MyClass {}
let obj = new MyClass();
obj instanceof MyClass;        // true
```

### 3.2 Prototype Chain

```javascript
// instanceof checks the entire prototype chain
[] instanceof Array;    // true
[] instanceof Object;  // true (Array extends Object)

function Parent() {}
function Child() {}
Child.prototype = Object.create(Parent.prototype);

let child = new Child();
child instanceof Child;   // true
child instanceof Parent;  // true
child instanceof Object;  // true
```

### 3.3 Limitations

```javascript
// Doesn't work with primitives
"hello" instanceof String;  // false (primitive, not object)
new String("hello") instanceof String;  // true

// Doesn't work across frames/realms
// Different window/frame contexts have different Array constructors
```

## 5. Object.prototype.toString()

### 5.1 The Most Reliable Method

```javascript
Object.prototype.toString.call("hello");     // "[object String]"
Object.prototype.toString.call(42);         // "[object Number]"
Object.prototype.toString.call(true);       // "[object Boolean]"
Object.prototype.toString.call(null);       // "[object Null]"
Object.prototype.toString.call(undefined);  // "[object Undefined]"
Object.prototype.toString.call([]);         // "[object Array]"
Object.prototype.toString.call({});        // "[object Object]"
Object.prototype.toString.call(new Date()); // "[object Date]"
Object.prototype.toString.call(/regex/);     // "[object RegExp]"
Object.prototype.toString.call(function(){}); // "[object Function]"
```

### 5.2 Helper Function

```javascript
function getType(value) {
  return Object.prototype.toString.call(value)
    .slice(8, -1)
    .toLowerCase();
}

getType([]);        // "array"
getType(null);      // "null"
getType(new Date()); // "date"
```

## 6. Type Guard Patterns

### 6.1 Primitive Type Guards

```javascript
function isString(value) {
  return typeof value === "string";
}

function isNumber(value) {
  return typeof value === "number" && !isNaN(value);
}

function isBoolean(value) {
  return typeof value === "boolean";
}

function isNull(value) {
  return value === null;
}

function isUndefined(value) {
  return typeof value === "undefined";
}

function isSymbol(value) {
  return typeof value === "symbol";
}

function isBigInt(value) {
  return typeof value === "bigint";
}
```

### 6.2 Object Type Guards

```javascript
function isArray(value) {
  return Array.isArray(value);
}

function isObject(value) {
  return value !== null && typeof value === "object";
}

function isPlainObject(value) {
  return (
    value !== null &&
    typeof value === "object" &&
    Object.prototype.toString.call(value) === "[object Object]" &&
    Object.getPrototypeOf(value) === Object.prototype
  );
}

function isDate(value) {
  return value instanceof Date;
}

function isRegExp(value) {
  return value instanceof RegExp;
}

function isFunction(value) {
  return typeof value === "function";
}
```

## 7. Comprehensive Type Detection

### 7.1 Universal Type Checker

```javascript
function getDetailedType(value) {
  // Handle null first (typeof quirk)
  if (value === null) return "null";
  
  // Primitives
  const primitiveType = typeof value;
  if (primitiveType !== "object") {
    if (primitiveType === "number" && isNaN(value)) {
      return "nan";
    }
    return primitiveType;
  }
  
  // Objects - use toString for reliability
  const tag = Object.prototype.toString.call(value).slice(8, -1);
  return tag.toLowerCase();
}

// Usage
getDetailedType(null);        // "null"
getDetailedType([]);          // "array"
getDetailedType({});          // "object"
getDetailedType(new Date());   // "date"
getDetailedType(NaN);         // "nan"
```

### 7.2 Type Validation Utility

```javascript
class TypeChecker {
  static is(value, type) {
    const actualType = getDetailedType(value);
    return actualType === type.toLowerCase();
  }
  
  static isOneOf(value, types) {
    return types.some(type => TypeChecker.is(value, type));
  }
  
  static validate(value, expectedType) {
    if (!TypeChecker.is(value, expectedType)) {
      throw new TypeError(
        `Expected ${expectedType}, got ${getDetailedType(value)}`
      );
    }
    return true;
  }
}

// Usage
TypeChecker.is([], "array");  // true
TypeChecker.isOneOf(42, ["number", "string"]);  // true
TypeChecker.validate("hello", "string");  // true
TypeChecker.validate(42, "string");  // TypeError
```

## 8. Real-World Patterns

### 8.1 Function Parameter Validation

```javascript
function processUser(user) {
  if (!isPlainObject(user)) {
    throw new TypeError("user must be an object");
  }
  if (!isString(user.name)) {
    throw new TypeError("user.name must be a string");
  }
  if (!isNumber(user.age) || user.age < 0) {
    throw new TypeError("user.age must be a non-negative number");
  }
  // Process user...
}
```

### 8.2 Duck Typing

```javascript
// Check for capabilities, not type
function canFly(obj) {
  return (
    typeof obj.fly === "function" &&
    typeof obj.wings === "number"
  );
}

function makeFly(obj) {
  if (canFly(obj)) {
    obj.fly();
  } else {
    throw new TypeError("Object cannot fly");
  }
}
```

### 8.3 Type-Safe Utilities

```javascript
function safeArrayOperation(arr, operation) {
  if (!Array.isArray(arr)) {
    throw new TypeError("First argument must be an array");
  }
  if (typeof operation !== "function") {
    throw new TypeError("Second argument must be a function");
  }
  return arr.map(operation);
}
```

## 10. Common Pitfalls & Best Practices

### Pitfall 1: Relying Only on typeof

```javascript
// ❌ Problematic
function process(data) {
  if (typeof data === "object") {
    // This includes null, arrays, dates, etc.
    data.process();  // Might fail
  }
}

// ✅ Better
function process(data) {
  if (isPlainObject(data)) {
    data.process();
  }
}
```

### Pitfall 2: instanceof with Primitives

```javascript
// ❌ Problematic
function isString(value) {
  return value instanceof String;  // false for primitives
}

// ✅ Better
function isString(value) {
  return typeof value === "string";
}
```

### Pitfall 3: Cross-Realm instanceof

```javascript
// ❌ Problematic (in iframes/workers)
let arr = [];
arr instanceof Array;  // Might be false in different realm

// ✅ Better
Array.isArray(arr);  // Works across realms
```

### Best Practice: Use the Right Tool

```javascript
// For primitives: typeof
typeof value === "string"

// For arrays: Array.isArray()
Array.isArray(value)

// For specific objects: instanceof
value instanceof Date

// For comprehensive checking: Object.prototype.toString()
Object.prototype.toString.call(value)

// For custom types: instanceof or duck typing
value instanceof MyClass
```
