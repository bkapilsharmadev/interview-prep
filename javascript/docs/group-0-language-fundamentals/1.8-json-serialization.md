# JSON Serialization & Structured Cloning

> "Moving data between contexts — understanding what survives the journey."

---

## 1. Concept Overview

JavaScript provides two main mechanisms for copying/serializing data:
- **JSON serialization**: `JSON.stringify()` / `JSON.parse()` — converts to/from JSON string
- **Structured cloning**: `structuredClone()` — deep copies objects (including non-JSON types)

Understanding these is crucial because:
- They have different capabilities and limitations
- They're used for data persistence, API communication, and copying
- Choosing the wrong one can lose data or cause errors
- They're common in real-world applications

## 2. JSON.stringify()

### 2.1 Basic Usage

```javascript
let obj = {
  name: "John",
  age: 30,
  active: true
};

JSON.stringify(obj);  // '{"name":"John","age":30,"active":true}'

// Pretty printing
JSON.stringify(obj, null, 2);
// {
//   "name": "John",
//   "age": 30,
//   "active": true
// }
```

### 2.2 What Gets Serialized

```javascript
// ✅ Serialized
JSON.stringify("string");     // '"string"'
JSON.stringify(42);           // '42'
JSON.stringify(true);         // 'true'
JSON.stringify(null);         // 'null'
JSON.stringify([1, 2, 3]);    // '[1,2,3]'
JSON.stringify({a: 1});       // '{"a":1}'

// ❌ Not serialized (omitted or become null)
JSON.stringify(undefined);    // undefined (not in object) or omitted
JSON.stringify(function(){}); // undefined (omitted)
JSON.stringify(Symbol());     // undefined (omitted)
JSON.stringify({a: undefined}); // '{}' (undefined properties omitted)
```

## 3. JSON.parse()

### 3.1 Basic Usage

```javascript
let json = '{"name":"John","age":30}';
let obj = JSON.parse(json);
// { name: "John", age: 30 }

// With reviver function
let obj2 = JSON.parse(json, (key, value) => {
  if (key === "age") return value * 2;
  return value;
});
// { name: "John", age: 60 }
```

### 3.2 Error Handling

```javascript
// Invalid JSON throws SyntaxError
try {
  JSON.parse("{ invalid }");
} catch (e) {
  console.error("Invalid JSON:", e.message);
}

// Safe parsing
function safeParse(json, defaultValue = null) {
  try {
    return JSON.parse(json);
  } catch {
    return defaultValue;
  }
}
```

## 4. Custom Serialization

### 4.1 toJSON() Method

```javascript
class User {
  constructor(name, birthDate) {
    this.name = name;
    this.birthDate = birthDate;
  }
  
  toJSON() {
    return {
      name: this.name,
      age: new Date().getFullYear() - this.birthDate.getFullYear()
    };
  }
}

let user = new User("John", new Date(1990, 0, 1));
JSON.stringify(user);  // '{"name":"John","age":34}'
```

### 4.2 Replacer Function

```javascript
let obj = {
  name: "John",
  password: "secret123",
  email: "john@example.com"
};

// Exclude sensitive fields
JSON.stringify(obj, (key, value) => {
  if (key === "password") return undefined;  // Omit
  return value;
});
// '{"name":"John","email":"john@example.com"}'

// Include only specific keys
JSON.stringify(obj, ["name", "email"]);
// '{"name":"John","email":"john@example.com"}'
```

## 5. Limitations of JSON

### 5.1 Data Types Not Supported

```javascript
let obj = {
  date: new Date(),           // Becomes string
  func: function(){},         // Omitted
  undefined: undefined,      // Omitted
  symbol: Symbol(),          // Omitted
  bigint: 123n,              // TypeError!
  regex: /test/,              // Becomes {}
  map: new Map(),             // Becomes {}
  set: new Set()              // Becomes {}
};

JSON.stringify(obj);
// '{"date":"2024-01-01T00:00:00.000Z"}'
```

### 5.2 Circular References

```javascript
let obj = { name: "John" };
obj.self = obj;  // Circular reference

JSON.stringify(obj);  // TypeError: Converting circular structure to JSON

// Solution: Handle in replacer
let seen = new WeakSet();
JSON.stringify(obj, (key, value) => {
  if (typeof value === "object" && value !== null) {
    if (seen.has(value)) {
      return "[Circular]";
    }
    seen.add(value);
  }
  return value;
});
```

## 6. structuredClone()

### 6.1 Basic Usage

```javascript
let obj = {
  name: "John",
  date: new Date(),
  map: new Map([["key", "value"]]),
  set: new Set([1, 2, 3])
};

let cloned = structuredClone(obj);
// Deep copy with support for more types than JSON
```

### 6.2 Supported Types

```javascript
// ✅ Supported by structuredClone
structuredClone(new Date());        // Date
structuredClone(new Map());        // Map
structuredClone(new Set());        // Set
structuredClone(new RegExp("test")); // RegExp
structuredClone(new ArrayBuffer(8)); // ArrayBuffer
structuredClone(new Uint8Array([1, 2])); // TypedArrays

// ❌ Not supported
structuredClone(function(){});     // Error
structuredClone(Symbol());         // Error
structuredClone({ get value(){} }); // Error (getters/setters)
```

## 7. Comparison: JSON vs structuredClone

| Feature | JSON.stringify/parse | structuredClone |
|---------|----------------------|-----------------|
| Functions | ❌ Omitted | ❌ Error |
| Dates | ✅ String | ✅ Date object |
| Maps/Sets | ❌ Becomes {} | ✅ Preserved |
| RegExp | ❌ Becomes {} | ✅ Preserved |
| Circular refs | ❌ Error | ✅ Handled |
| BigInt | ❌ Error | ❌ Error |
| Symbols | ❌ Omitted | ❌ Error |
| TypedArrays | ❌ Limited | ✅ Preserved |

## 8. Real-World Patterns

### 8.1 Deep Copying Objects

```javascript
// Using JSON (limited but simple)
function deepCopyJSON(obj) {
  return JSON.parse(JSON.stringify(obj));
}

// Using structuredClone (better support)
function deepCopy(obj) {
  return structuredClone(obj);
}

// Manual deep copy (most control)
function deepCopyManual(obj) {
  if (obj === null || typeof obj !== "object") return obj;
  if (obj instanceof Date) return new Date(obj);
  if (obj instanceof Array) return obj.map(item => deepCopyManual(item));
  if (typeof obj === "object") {
    let copy = {};
    for (let key in obj) {
      if (obj.hasOwnProperty(key)) {
        copy[key] = deepCopyManual(obj[key]);
      }
    }
    return copy;
  }
}
```

### 8.2 API Communication

```javascript
// Sending to API
async function sendData(data) {
  const response = await fetch("/api/data", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(data)
  });
  return response.json();
}

// Receiving from API
async function getData() {
  const response = await fetch("/api/data");
  const json = await response.json();
  // Convert date strings back to Date objects
  if (json.createdAt) {
    json.createdAt = new Date(json.createdAt);
  }
  return json;
}
```

### 8.3 Local Storage

```javascript
// Saving to localStorage
function saveToStorage(key, data) {
  try {
    localStorage.setItem(key, JSON.stringify(data));
  } catch (e) {
    console.error("Storage quota exceeded", e);
  }
}

// Loading from localStorage
function loadFromStorage(key, defaultValue = null) {
  try {
    const item = localStorage.getItem(key);
    return item ? JSON.parse(item) : defaultValue;
  } catch (e) {
    console.error("Invalid JSON in storage", e);
    return defaultValue;
  }
}
```

## 10. Common Pitfalls & Best Practices

### Pitfall 1: Losing Data Types

```javascript
// ❌ Problematic
let data = { date: new Date() };
let stored = JSON.stringify(data);
let restored = JSON.parse(stored);
console.log(restored.date);  // String, not Date!

// ✅ Better
let data = { date: new Date() };
let stored = JSON.stringify(data, (key, value) => {
  if (value instanceof Date) {
    return { __type: "Date", value: value.toISOString() };
  }
  return value;
});

let restored = JSON.parse(stored, (key, value) => {
  if (value && value.__type === "Date") {
    return new Date(value.value);
  }
  return value;
});
```

### Pitfall 2: Circular References

```javascript
// ❌ Problematic
let obj = { name: "John" };
obj.self = obj;
JSON.stringify(obj);  // TypeError

// ✅ Better: Use structuredClone or handle in replacer
let cloned = structuredClone(obj);  // Works!
```

### Pitfall 3: BigInt Serialization

```javascript
// ❌ Problematic
let data = { id: 123n };
JSON.stringify(data);  // TypeError

// ✅ Better: Convert to string
JSON.stringify(data, (key, value) =>
  typeof value === 'bigint' ? value.toString() : value
);
```

### Best Practice: Choose the Right Tool

```javascript
// Use JSON for:
// - API communication
// - LocalStorage
// - Simple data structures
// - When you need string representation

// Use structuredClone for:
// - Deep copying complex objects
// - Preserving Date, Map, Set, etc.
// - When you need object, not string
// - Modern environments only
```
